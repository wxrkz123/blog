# 趋势交易策略分析

## MACD indicator trading strategy.py

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import pandas_ta as ta
from matplotlib.pyplot import MultipleLocator
from docx import Document

def calculate_max_drawdown(cumulative_returns):
    # 通过累计回报计算资金曲线，取1加上回报序列即可
    portfolio_values = 1 + cumulative_returns
    # 计算累计最高值
    running_max = portfolio_values.cummax()
    # 计算回撤
    drawdowns = 1 - portfolio_values / running_max
    # 计算最大回撤
    max_drawdown = drawdowns.max()

    return max_drawdown

# 从excel中加载数据
data = pd.read_excel('E:\\MZL.DCE豆粕主连.xlsx')
# 对时间进行格式化
data['Date'] = pd.to_datetime(data['Date'])
# 计算MACD指标
data['MACD'], data['MACD_Signal'], data['MACD_Hist'] = ta.macd(data['Close'])

# 计算EMA12、EMA26、DIF、DEA、以及MACD
short_period = 12
long_period = 26
dea_period = 5

data['EMA12'] = data['Close'].ewm(span=short_period).mean()
data['EMA26'] = data['Close'].ewm(span=long_period).mean()
data['DIF'] = data['EMA12'] - data['EMA26']
data['DEA'] = data['DIF'].ewm(span=dea_period).mean()
data['MACD'] = 2 * (data['DIF'] - data['DEA'])

# 初始化仓位
position = 0
buy_signals, sell_signals = [], []
trades = []

# 更新买入和卖出信号, 执行交易逻辑
data['Buy_Signal_Flag'] = 0
data['Sell_Signal_Flag'] = 0

for i in range(1, len(data)):
    row_today = data.iloc[i]
    row_yesterday = data.iloc[i - 1]

    # 当价格突破上轨，买入开仓
    if (row_yesterday['MACD'] > 0) and (row_yesterday['DIF'] > row_yesterday['DEA']):
        if position != 1:
            data.loc[row_today.name, 'Buy_Signal_Flag'] = 1
            buy_signals.append((i, row_yesterday['Close']))
            trades.append({'Trade_Type':'Buy', 'type':'Open', 'Price': row_yesterday['Close'], 'Date':row_today.name})
            position = 1
            data.loc[row_today.name, 'Return'] = (row_today['Close'] - row_yesterday['Close']) / row_today['Close']

    # 当价格跌穿下轨，卖出开仓
    if (row_yesterday['MACD'] < 0) and (row_yesterday['DIF'] < row_yesterday['DEA']):
        if position != -1:
            data.loc[row_today.name, 'Sell_Signal_Flag'] = -1
            sell_signals.append((i, row_yesterday['Close']))
            trades.append({'Trade_Type':'Sell', 'type':'Open', 'Price': row_yesterday['Close'], 'Date':row_today.name})
            position = -1
            data.loc[row_today.name, 'Return'] = (row_yesterday['Close'] - row_today['Close']) / row_today['Close']
            
    # 市场收盘，清空仓位
    if position != 0:
        if position == 1:
            sell_signals.append((i, row_today['Close']))
            trades.append({'Trade_Type':'Sell', 'type':'Close', 'Price':row_today['Close'], 'Date':row_today.name})
        if position == -1:
            buy_signals.append((i, row_today['Close']))
            trades.append({'Trade_Type':'Buy', 'type':'Close', 'Price':row_today['Close'], 'Date':row_today.name})
        position = 0

# 根据清理后的数据计算累计收益率
data['Cumulative_Return'] = (1 + data['Return']).cumprod() - 1

# 创建DataFrame
buy_signals_df = pd.DataFrame(buy_signals, columns=['Index', 'Price'])
sell_signals_df = pd.DataFrame(sell_signals, columns=['Index', 'Price'])
trades_df = pd.DataFrame(trades)

# 计算置信区间
data['Confidence_Interval_Upper_Track'] = data['Close'] + 1.96 * np.std(data['Close'])
data['Confidence_Interval_Lower_Track'] = data['Close'] - 1.96 * np.std(data['Close'])

#找出计算MACD时的买卖信号点
data['Buy_Signal'] = np.where((data['MACD'] > 0) & (data['DIF'] > data['DEA']), data['Close'], np.nan)
data['Sell_Signal'] = np.where((data['MACD'] < 0) & (data['DIF'] < data['DEA']), data['Close'], np.nan)

# 绘图
fig, (ax1, ax2) = plt.subplots(2, sharex=True, figsize=(12,8))

ax1.plot(data['Close'], label='Close', color='blue')
ax1.fill_between(data.index,data['Confidence_Interval_Lower_Track'], data['Confidence_Interval_Upper_Track'], color='gray', alpha=0.3)
ax1.plot(data.index, data['Buy_Signal'], marker='^', markersize=12, lw=0, color='g', label='buy')
ax1.plot(data.index, data['Sell_Signal'], marker='v', markersize=12, lw=0, color='r', label='sell')
ax1.legend(loc='upper left')
ax1.set_title('Close and Confidence Interval')

ax2.plot(data['DIF'], label='DIF', color='orange')
ax2.plot(data['DEA'], label='DEA', color='purple')
ax2.bar(data.index, data['MACD'], label='MACD', color='gray', alpha=0.3)
ax2.xaxis.set_major_locator(MultipleLocator(250)) 
ax2.legend(loc='upper left')
ax2.set_title('MACD Indicator')

plt.tight_layout()
plt.savefig('Close_Price_and_MACD_Analysis.png')
plt.show()

# 确认买入卖出时机
buy_signals = [index for index, value in zip(data.index, data['Buy_Signal_Flag']) if value == 1]
sell_signals = [index for index, value in zip(data.index, data['Sell_Signal_Flag']) if value == -1]

if len(buy_signals) == 0 or len(sell_signals) == 0:
    print("No enough buy and sell signals generated. Please check your data and strategy.")
else:
    # 确保卖出在买入之后，买入在卖出之前
    buy_signals_filtered = [idx for idx in buy_signals if idx < max(sell_signals)]
    sell_signals_filtered = [idx for idx in sell_signals if idx > min(buy_signals_filtered)]

    # 计算交易次数（每次买入信号到紧邻的卖出信号算一次交易）
    trade_count = min(len(buy_signals_filtered), len(sell_signals_filtered))

    # 计算盈利次数
    profit_times = 0
    for i in range(trade_count):
        # 找到当前买入点之后的第一个卖出点
        buy_price = data['Close'][buy_signals_filtered[i]]
        sell_price = data['Close'][sell_signals_filtered[i]]
        if sell_price > buy_price:
            # 如果卖出价高于买入价，说明这次交易盈利，投资成功
            profit_times += 1

    # 计算胜率
    win_rate = profit_times / trade_count

# 确保数据都是有效的
data = data[~data['Return'].isna()]
# 确保数据都是有效的
data = data.dropna(subset=['Return'])
# 使用上述函数计算最大回撤
max_drawdown = calculate_max_drawdown(data['Cumulative_Return'])

# 生成累计收益率图
plt.figure(figsize=(12, 6))
plt.plot(data.index, data['Cumulative_Return'], label='Cumulative Return', color='green')
plt.title('Cumulative Return')
plt.xlabel('Date')
plt.ylabel('Cumulative Return')
plt.legend()
plt.savefig('MACD_indicator_trading_strategy_Cumulative_Return_Analysis.png')
plt.show()

# 计算夏普比率
risk_free_rate = 0.03  # 年无风险利率为3%
daily_return = data['Return']  # 每日收益率
daily_std = daily_return.std()  # 每日收益率的标准差
average_daily_return = daily_return.mean()  # 每日收益率的平均值

# 将年无风险利率转换为每日无风险利率
daily_risk_free_rate = (1 + risk_free_rate) ** (1/360) - 1  # 假设一年有360个交易日

# 计算夏普比率
sharpe_ratio = (average_daily_return - daily_risk_free_rate) / daily_std

# 创建一个文档对象
doc = Document()

# 将业绩归因的指标添加到文档中
doc.add_heading('MACD indicator trading strategy')
doc.add_paragraph(f"Total number of trades: {trade_count}")
doc.add_paragraph(f"Total number of profitable trades: {profit_times}")
doc.add_paragraph(f"Win rate: {win_rate*100}%")
doc.add_paragraph("The maximum drawdown : {:.2%}".format(max_drawdown))
doc.add_paragraph(f"Sharpe Ratio: {sharpe_ratio}")

# 保存文档
doc.save("MACD_indicator_trading_strategy.docx")

# 创建一个ExcelWriter对象
with pd.ExcelWriter('MACD_indicator_trading_strategy_output_data.xlsx') as writer:
    data.to_excel(writer, sheet_name='Original Data')
    buy_signals_df.to_excel(writer, sheet_name='Buy Signals', index=False)
    sell_signals_df.to_excel(writer, sheet_name='Sell Signals', index=False)
    trades_df.to_excel(writer, sheet_name='Trades', index=False)
```

```python
import pandas as pd
import matplotlib.pyplot as plt
from docx import Document

def calculate_max_drawdown(cumulative_returns):
    portfolio_values = 1 + cumulative_returns
    running_max = portfolio_values.cummax()
    drawdowns = 1 - portfolio_values / running_max
    max_drawdown = drawdowns.max()
    return max_drawdown

# 从excel中加载数据
data = pd.read_excel('E:\\MZL.DCE豆粕主连.xlsx')

# 对时间进行格式化
data['Date'] = pd.to_datetime(data['Date'])

# Dual Thrust策略
N = 5 
K1 = 0.5 
K2 = 0.5
data['HH'] = data['High'].rolling(window=N).max()
data['LC'] = data['Close'].rolling(window=N).min()
data['HC'] = data['Close'].rolling(window=N).max()
data['LL'] = data['Low'].rolling(window=N).min()

# 计算Range
data['Range'] = data[['HH', 'LC', 'HC', 'LL']].apply(lambda x: max(x['HH'] - x['LC'], x['HC'] - x['LL']), axis=1)
data['BuyLine'] = data['Open'] + K1 * data['Range']
data['SellLine'] = data['Open'] - K2 * data['Range']

# 初始化仓位
position = 0
buy_signals, sell_signals = [], []
trades = []

# 更新买入和卖出信号, 执行交易逻辑
data['Buy_Signal_Flag'] = 0
data['Sell_Signal_Flag'] = 0
data['Return'] = 0

for i in range(N, len(data)):
    row_today = data.iloc[i]
    row_yesterday = data.iloc[i - 1]

    if row_today['High'] > row_yesterday['BuyLine'] and row_today['Low'] <= row_yesterday['BuyLine']:
        if position <= 0 and row_today['Open'] > row_yesterday['BuyLine']:
            data.loc[row_today.name, 'Buy_Signal_Flag'] = 1
            buy_signals.append((i, row_yesterday['BuyLine']))
            position += 1 
            trades.append({'Trade_Type':'Buy', 'Price':row_yesterday['BuyLine'], 'Date':row_today.name})
            data.loc[row_today.name, 'Return'] = (row_today['Close'] - row_yesterday['BuyLine']) / row_yesterday['BuyLine']

    if row_today['Low'] < row_yesterday['SellLine'] and row_today['High'] >= row_yesterday['SellLine']:
        if position >= 0 and row_today['Open'] < row_yesterday['SellLine']:
            data.loc[row_today.name, 'Sell_Signal_Flag'] = -1
            sell_signals.append((i, row_yesterday['SellLine']))
            position -= 1
            trades.append({'Trade_Type':'Sell', 'Price':row_yesterday['SellLine'], 'Date':row_today.name})
            data.loc[row_today.name, 'Return'] = (row_yesterday['SellLine'] - row_today['Close']) / row_yesterday['SellLine']
    
    # 市场收盘，清空仓位
    if position != 0:
        if position == 1:
            sell_signals.append((i, row_today['Close']))
            trades.append({'Trade_Type':'Sell', 'Price':row_today['Close'], 'Date':row_today.name})
        if position == -1:
            buy_signals.append((i, row_today['Close']))
            trades.append({'Trade_Type':'Buy', 'Price':row_today['Close'], 'Date':row_today.name})
        position = 0

# 根据清理后的数据计算累计收益率
data['Cumulative_Return'] = (1 + data['Return']).cumprod() - 1

# 创建DataFrame
buy_signals_df = pd.DataFrame(buy_signals, columns=['Index', 'Price'])
sell_signals_df = pd.DataFrame(sell_signals, columns=['Index', 'Price'])
trades_df = pd.DataFrame(trades)

# 修改确保所有没有买卖操作的交易日收益为 0
data.loc[(data['Buy_Signal_Flag'] == 0) & (data['Sell_Signal_Flag'] == 0), 'Return'] = 0

# 画出价格和突破点
plt.figure(figsize=(12,8))
plt.plot(data.index, data['Close'], label='Close Price', color='blue')
plt.scatter([trade['Date'] for trade in trades if trade['Trade_Type']=='Buy'], trades_df[trades_df['Trade_Type']=='Buy']['Price'], color='green',label='Buy Signal', marker='^', alpha=1)
plt.scatter([trade['Date'] for trade in trades if trade['Trade_Type']=='Sell'], trades_df[trades_df['Trade_Type']=='Sell']['Price'], color='red',label='Sell Signal', marker='v', alpha=1)
plt.title('Prices and Buy and Sell Signals of Dual Thrust Strategy')
plt.xlabel('Date')
plt.ylabel('Price')
plt.legend()
plt.savefig('Prices_and_Buy_and_Sell_Signals_of_Dual_Thrust_Strategy.png')
plt.show()

# 确定交易买卖点
buy_signals = [index for index, value in zip(data.index, data['Buy_Signal_Flag']) if value == 1]
sell_signals = [index for index, value in zip(data.index, data['Sell_Signal_Flag']) if value == -1]
```

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from docx import Document

# ATR计算函数
def calculate_atr(data: pd.DataFrame, period: int=14) -> pd.Series:
    high = data['High']
    low = data['Low']
    close = data['Close']
    
    high_low = high - low
    high_close = np.abs(high - close.shift())
    low_close = np.abs(low - close.shift())
    
    ranges = pd.concat([high_low, high_close, low_close], axis=1)
    true_range = ranges.max(axis=1)
    atr = true_range.rolling(period).mean()
    return atr

def calculate_max_drawdown(cumulative_returns):
    portfolio_values = 1 + cumulative_returns
    running_max = portfolio_values.cummax()
    drawdowns = 1 - portfolio_values / running_max
    max_drawdown = drawdowns.max()
    return max_drawdown

# 从excel中加载数据
data = pd.read_excel('E:\\MZL.DCE豆粕主连.xlsx')
# 对时间进行格式化
data['Date'] = pd.to_datetime(data['Date'])

# ATR策略
N = 10
M = 1
# 计算ATR
data['ATR'] = calculate_atr(data, N)

# 计算上下轨
data['Upper_Track'] = data['Open'] + M * data['ATR']
data['Lower_Track'] = data['Open'] - M * data['ATR']

# 初始化仓位
position = 0
buy_signals, sell_signals = [], []
trades = []
data['Return'] = 0

for i in range(N, len(data)):
    row_today = data.iloc[i]
    row_yesterday = data.iloc[i-1]

    # 回到市场收盘
    if row_today['High'] > row_yesterday['Upper_Track'] and row_today['Low'] <= row_yesterday['Upper_Track']:
        if position != 1 and row_today['Open'] > row_yesterday['Upper_Track']:
            data.loc[row_today.name, 'Buy_Signal_Flag'] = 1
            buy_signals.append((i, row_yesterday['Upper_Track']))
            position = 1 
            trades.append({'Trade_Type':'Buy', 'Price':row_yesterday['Upper_Track'], 'Date':row_today.name})
            data.loc[row_today.name, 'Return'] = (row_today['Close'] - row_yesterday['Upper_Track']) / row_yesterday['Upper_Track']

    # 进入市场开盘
    if row_today['Low'] < row_yesterday['Lower_Track'] and row_today['High'] >= row_yesterday['Lower_Track']:
        if position != -1 and row_today['Open'] < row_yesterday['Upper_Track']:
            data.loc[row_today.name, 'Sell_Signal_Flag'] = -1
            sell_signals.append((i, row_yesterday['Lower_Track']))
            position = -1
            trades.append({'Trade_Type':'Sell', 'Price':row_yesterday['Lower_Track'], 'Date':row_today.name})
            data.loc[row_today.name, 'Return'] = (row_yesterday['Lower_Track'] - row_today['Close']) / row_yesterday['Lower_Track']

    # 市场收盘，清空仓位
    if position != 0:
        if position == 1:
            data.loc[row_today.name, 'Sell_Signal_Flag'] = -1
            sell_signals.append((i, row_today['Close']))
            trades.append({'Trade_Type':'Sell', 'Price':row_today['Close'], 'Date':row_today.name})
        if position == -1:
            data.loc[row_today.name, 'Buy_Signal_Flag'] = 1
            buy_signals.append((i, row_today['Close']))
            trades.append({'Trade_Type':'Buy', 'Price':row_today['Close'], 'Date':row_today.name})
        position = 0

# 根据清理后的数据计算累计收益率
data['Cumulative_Return'] = (1 + data['Return']).cumprod() - 1

# 创建DataFrame
buy_signals_df = pd.DataFrame(buy_signals, columns=['Index', 'Price'])
sell_signals_df = pd.DataFrame(sell_signals, columns=['Index', 'Price'])
trades_df = pd.DataFrame(trades)

plt.plot(data.index, data['Close'], label='Close Price', color='blue')
plt.plot(data.index, data['Upper_Track'], label='Upper Track', color='red')
plt.plot(data.index, data['Lower_Track'], label='Lower Track', color='green')
plt.scatter([trade['Date'] for trade in trades if trade['Trade_Type']=='Buy'], trades_df[trades_df['Trade_Type']=='Buy']['Price'], color='green',label='Buy Signal', marker='^', alpha=1)
plt.scatter([trade['Date'] for trade in trades if trade['Trade_Type']=='Sell'], trades_df[trades_df['Trade_Type']=='Sell']['Price'], color='red',label='Sell Signal', marker='v', alpha=1)
plt.title('Prices and Buy and Sell Signals of ATR Strategy')
plt.xlabel('Date')
plt.ylabel('Price')
plt.legend()
plt.savefig('Prices_and_Buy_and_Sell_Signals_of_ATR_Strategy.png')
plt.show()

# 确认买入卖出时机
buy_signals = [index for index, value in zip(data.index, data['Buy_Signal_Flag']) if value == 1]
sell_signals = [index for index, value in zip(data.index, data['Sell_Signal_Flag']) if value == -1]

if len(buy_signals) == 0 or len(sell_signals) == 0:
    print("No enough buy and sell signals generated. Please check your data and strategy.")
else:
    # 确保卖出在买入之后，买入在卖出之前
    buy_signals_filtered = [idx for idx in buy_signals if idx < max(sell_signals)]
    sell_signals_filtered = [idx for idx in sell_signals if idx > min(buy_signals_filtered)]

    # 计算交易次数（每次买入信号到紧邻的卖出信号算一次交易）
    trade_count = min(len(buy_signals_filtered), len(sell_signals_filtered))

    # 计算盈利次数
    profit_times = 0
    for i in range(trade_count):
        buy_price = data['Close'][buy_signals_filtered[i]]
        sell_price = data['Close'][sell_signals_filtered[i]]
        if sell_price > buy_price:
            profit_times += 1

    # 计算胜率
    win_rate = profit_times / trade_count

# 确保数据都是有效的
data = data[~data['Return'].isna()]

max_drawdown = calculate_max_drawdown(data['Cumulative_Return'])

plt.figure(figsize=(12, 6))
plt.plot(data.index, data['Cumulative_Return'], label='Cumulative Return', color='green')
plt.title('Cumulative Return')
plt.xlabel('Date')
plt.ylabel('Cumulative Return')
plt.legend()
plt.savefig('ATR_Strategy_Cumulative_Return_Analysis.png')
plt.show()

# 计算夏普比率
risk_free_rate = 0.03  # 年无风险利率为3%
daily_return = data['Return']  # 每日收益率
daily_std = daily_return.std()  # 每日收益率的标准差
average_daily_return = daily_return.mean()  # 每日收益率的平均值

# 将年无风险利率转换为每日无风险利率
daily_risk_free_rate = (1 + risk_free_rate) ** (1/360) - 1  # 假设一年有360个交易日

sharpe_ratio = (average_daily_return - daily_risk_free_rate) / daily_std
```
